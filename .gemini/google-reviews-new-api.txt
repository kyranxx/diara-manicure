  // Google Reviews Fetching with NEW Places API (not deprecated)
  useEffect(() => {
    const fetchGoogleReviews = async () => {
      try {
        // Check if Google Maps API is available
        if (!(window as any).google?.maps?.importLibrary) {
          console.warn('⚠️ Google Maps API not yet loaded');
          return;
        }

        console.log('🔄 Fetching Google Reviews with NEW Places API...');

        // Import the new Places library
        const { Place } = await (window as any).google.maps.importLibrary("places");

        // Search for the place using the new API
        const request = {
          textQuery: 'Diara Manicure Trnava',
          fields: ['displayName', 'reviews', 'id'],
          locationBias: { lat: 48.3774, lng: 17.5851 }, // Trnava coordinates
        };

        const { places } = await Place.searchByText(request);

        if (places && places.length > 0) {
          const place = places[0];
          
          // Fetch reviews field explicitly
          await place.fetchFields({ fields: ['reviews'] });

          if (place.reviews) {
            console.log('✅ Google Reviews loaded:', place.reviews.length);
            
            const googleReviews: Testimonial[] = place.reviews
              .filter((review: any) => review.rating && review.rating >= 4 && review.text && review.text.trim() !== '')
              .slice(0, 5)
              .map((review: any) => ({
                text: review.text || '',
                author: review.authorAttribution?.displayName || 'Anonymous',
                photo: review.authorAttribution?.photoUri || null,
                rating: review.rating,
                source: 'google' as const
              }));

            setTestimonials(prev => {
              // Avoid duplicates based on author name
              const newReviews = googleReviews.filter(gr => !prev.some(pr => pr.author === gr.author));
              // Shuffle combined reviews
              const combined = [...newReviews, ...prev];
              for (let i = combined.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [combined[i], combined[j]] = [combined[j], combined[i]];
              }
              return combined;
            });
          }
        }
      } catch (error) {
        console.error('❌ Error fetching Google Reviews:', error);
      }
    };

    // Wait for Google Maps API with optimized polling
    if ((window as any).google?.maps?.importLibrary) {
      fetchGoogleReviews();
    } else {
      let attempts = 0;
      const maxAttempts = 30; // 30 attempts = ~15 seconds max wait
      const checkInterval = setInterval(() => {
        attempts++;
        if ((window as any).google?.maps?.importLibrary) {
          clearInterval(checkInterval);
          console.log('✅ Google Maps API ready, fetching reviews...');
          fetchGoogleReviews();
        } else if (attempts >= maxAttempts) {
          clearInterval(checkInterval);
          console.warn('⏱️ Google Maps API timed out - reviews will not load');
        }
      }, 500); // Check every 500ms (less aggressive than before)

      return () => clearInterval(checkInterval);
    }
  }, []);
